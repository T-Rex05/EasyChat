```python
class ArtifactStore:
    """Artifact 存储，复用 AgentForge 的记忆系统"""

    def __init__(self, user_id: int, thread_id: str):
        self.user_id = user_id
        self.thread_id = thread_id
        self.memory_service = get_memory_service()

    def store_artifact(self, name: str, value: Any, metadata: dict = None):
        """存储 Artifact 到记忆系统"""
        # 使用 Semantic Memory 存储
        self.memory_service.add_semantic_memory(
            user_id=self.user_id,
            thread_id=self.thread_id,
            content=json.dumps(value),
            metadata={"artifact_name": name, **metadata}
        )

    def get_artifact(self, name: str) -> Optional[Any]:
        """从记忆系统读取 Artifact"""
        # 使用 GraphRAG 检索
        result = self.memory_service.retrieve(
            user_id=self.user_id,
            query=f"artifact:{name}",
            mode="semantic"
        )
        return json.loads(result) if result else None
```

### 2.2 ⚠️ 中等难度部分

#### (1) DAG 调度器

**问题**：Sea-Mult-Agent 有独立的 DAG Scheduler，Hermes 是单 Agent 循环

**解决方案 1**：在 Hermes 内部实现轻量 DAG

```python
# backend/hermes/agent/dag_executor.py

class DAGExecutor:
    """轻量 DAG 执行器，嵌入 Hermes Agent"""

    def __init__(self, agent: AIAgent):
        self.agent = agent
        self.nodes = {}  # TaskNode
        self.artifacts = {}  # Artifact

    def execute_dag(self, dag: PlanGraph) -> AsyncGenerator[Event, None]:
        """执行 DAG 并流式返回事件"""
        # 1. 拓扑排序
        ready_nodes = self._get_ready_nodes()

        # 2. 并发执行（最多 2 个）
        with ThreadPoolExecutor(max_workers=2) as executor:
            futures = []
            for node in ready_nodes:
                if len(futures) >= 2:
                    # 等待其中一个完成
                    done, _ = wait(futures, return_when=FIRST_COMPLETED)
                    for future in done:
                        yield from self._process_result(future)

                # 提交任务
                future = executor.submit(self._execute_node, node)
                futures.append(future)

            # 等待所有完成
            for future in futures:
                yield from self._process_result(future)

    def _execute_node(self, node: TaskNode) -> Event:
        """执行单个节点"""
        # 1. 收集上游 Artifact
        inputs = {name: self.artifacts[name] for name in node.required_artifacts}

        # 2. 调用 Hermes 工具
        tool_name = self._get_tool_for_node(node)
        result = self.agent.call_tool(tool_name, inputs)

        # 3. 存储 Artifact
        for artifact_name in node.output_artifacts:
            self.artifacts[artifact_name] = result[artifact_name]

        # 4. 返回事件
        return Event(
            type="task_completed",
            node_id=node.id,
            artifacts=node.output_artifacts
        )
```

#### (2) 执行环境隔离

**问题**：Sea-Mult-Agent 有独立的 Docker Sandbox 服务，Hermes 的 terminal_tool 在宿主机执行

**解决方案**：扩展 terminal_tool 支持 Docker 容器

```python
# backend/hermes/tools/docker_sandbox_tool.py

from tools.terminal_tool import TerminalTool
import docker

class DockerSandboxTool(TerminalTool):
    """Docker 沙箱工具，继承 terminal_tool"""

    def __init__(self):
        self.client = docker.from_env()
        self.containers = {}  # runtime_id -> Container

    def create_runtime(self, image: str, resources: dict) -> str:
        """创建隔离运行时"""
        container = self.client.containers.run(
            image=image,
            detach=True,
            cpu_quota=resources.get("cpu", 2) * 100000,
            mem_limit=resources.get("memory", "4g"),
            pids_limit=resources.get("pid", 256),
            network_mode=resources.get("network", "bridge"),
            volumes={
                self.workspace_path: {"bind": "/workspace", "mode": "rw"}
            }
        )
        runtime_id = container.id
        self.containers[runtime_id] = container
        return runtime_id

    def execute(self, runtime_id: str, command: str) -> Tuple[int, str, str]:
        """在容器中执行命令"""
        container = self.containers[runtime_id]
        exit_code, output = container.exec_run(command)
        return exit_code, output, ""

    def destroy_runtime(self, runtime_id: str):
        """销毁容器"""
        container = self.containers[runtime_id]
        container.remove(force=True)
        del self.containers[runtime_id]
```

**集成方式**：
- 在 `backend/tools/tools_decorator.py` 中注册 `docker_sandbox_tool`
- 通过权限系统控制访问

```typescript
// agent_frontend/src/components/DAGViewer.tsx

import ReactFlow, { Node, Edge } from 'reactflow';

interface DAGViewerProps {
  plan: PlanGraph;
  events: Event[];
}

export function DAGViewer({ plan, events }: DAGViewerProps) {
  // 转换为 React Flow 格式
  const nodes: Node[] = Object.values(plan.nodes).map(node => ({
    id: node.id,
    type: 'taskNode',
    data: {
      label: node.type,
      status: getNodeStatus(node, events),
    },
    position: { x: node.x, y: node.y },
  }));

  const edges: Edge[] = Object.values(plan.nodes).flatMap(node =>
    node.dependencies.map(dep => ({
      id: `${dep}-${node.id}`,
      source: dep,
      target: node.id,
    }))
  );

  return (
    <div style={{ width: '100%', height: '600px' }}>
      <ReactFlow nodes={nodes} edges={edges} />
    </div>
  );
}
```

**问题**：Sea-Mult-Agent 有独立的事件流，AgentForge 只有对话流

```python
# backend/controller/chat/stream.py

async def token_stream_events(
    message: str,
    thread_id: str,
    user_id: int,
) -> AsyncGenerator[str, None]:
    """扩展 SSE，支持 DAG 事件"""

    # 1. 判断是否是论文复现任务
    if is_paper_reproduction(message):
        # 2. 创建 DAG
        plan = await create_paper_reproduction_plan(message, user_id)

        # 3. 推送 DAG 创建事件
        yield f"event: dag_created\ndata: {json.dumps(plan)}\n\n"

        # 4. 执行 DAG
        async for event in execute_dag(plan):
            yield f"event: {event.type}\ndata: {json.dumps(event)}\n\n"

        # 5. 推送完成事件
        yield f"event: dag_completed\ndata: {json.dumps(plan)}\n\n"
    else:
        # 原有对话流程
        async for token in hermes_stream(message, thread_id, user_id):
            yield f"event: token\ndata: {token}\n\n"
```
